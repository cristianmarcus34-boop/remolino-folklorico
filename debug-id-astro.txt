    <script define:vars={{ id }}>
        const form = document.querySelector<HTMLFormElement>("#evento-form");
        const errorMsg = document.getElementById("error-msg");
        const submitBtn = document.getElementById("submit-btn");
        const imagenFile = document.getElementById("imagenFile");
        const preview = document.getElementById("preview");
        const previewContainer = document.getElementById("preview-container");

        const tabs = document.querySelectorAll(".imagen-tab");
        const panels = document.querySelectorAll(".imagen-panel");

        tabs.forEach((tab) => {
            tab.addEventListener("click", () => {
                const target = tab.getAttribute("data-tab");
                tabs.forEach((t) => t.classList.remove("active"));
                tab.classList.add("active");
                panels.forEach((p) => {
                    p.classList.toggle(
                        "active",
                        p.getAttribute("data-panel") === target,
                    );
                });
            });
        });

        if (imagenFile && "files" in imagenFile) {
            imagenFile.addEventListener("change", () => {
                const file = imagenFile.files && imagenFile.files[0];
                if (!file) {
                    if (previewContainer && "style" in previewContainer) {
                        previewContainer.style.display = "none";
                    }
                    return;
                }
                const reader = new FileReader();
                reader.onload = (e) => {
                    if (preview && "src" in preview) {
                        preview.src = String(e.target ? e.target.result : "");
                    }
                    if (previewContainer && "style" in previewContainer) {
                        previewContainer.style.display = "block";
                    }
                };
                reader.readAsDataURL(file);
            });
        }

        if (form && errorMsg && submitBtn) {
            form.addEventListener("submit", async (e) => {
                e.preventDefault();
                errorMsg.style.display = "none";
                submitBtn.setAttribute("disabled", "true");
                submitBtn.textContent = "Guardando...";

                const elTitulo = document.getElementById("titulo");
                const elLugar = document.getElementById("lugar");
                const elFecha = document.getElementById("fecha");
                const elHora = document.getElementById("hora");
                const elImagenUrl = document.getElementById("imagenUrl");

                const titulo =
                    elTitulo &&
                    "value" in elTitulo &&
                    typeof elTitulo.value === "string"
                        ? elTitulo.value
                        : "";
                const lugar =
                    elLugar &&
                    "value" in elLugar &&
                    typeof elLugar.value === "string"
                        ? elLugar.value
                        : "";
                const fecha =
                    elFecha &&
                    "value" in elFecha &&
                    typeof elFecha.value === "string"
                        ? elFecha.value
                        : "";
                const hora =
                    elHora &&
                    "value" in elHora &&
                    typeof elHora.value === "string"
                        ? elHora.value
                        : "";
                let imagenUrl =
                    elImagenUrl &&
                    "value" in elImagenUrl &&
                    typeof elImagenUrl.value === "string"
                        ? elImagenUrl.value
                        : "";

                console.log("Payload inicial:", {
                    titulo,
                    lugar,
                    fecha,
                    hora,
                    imagenUrl,
                });

                try {
                    const file =
                        imagenFile && "files" in imagenFile && imagenFile.files
                            ? imagenFile.files[0]
                            : null;
                    if (file) {
                        console.log("Subiendo archivo:", file.name);
                        const uploadData = new FormData();
                        uploadData.append("file", file);
                        const uploadRes = await fetch("/api/upload", {
                            method: "POST",
                            body: uploadData,
                            credentials: "include",
                        });
                        if (!uploadRes.ok) {
                            const data = await uploadRes.json();
                            throw new Error(
                                data.error || "Error al subir la imagen",
                            );
                        }
                        const { url } = await uploadRes.json();
                        console.log("Imagen subida:", url);
                        imagenUrl = url;

                        const alt = titulo || "Imagen de evento";
                        const galeriaRes = await fetch("/api/galeria", {
                            method: "POST",
                            headers: { "Content-Type": "application/json" },
                            credentials: "include",
                            body: JSON.stringify({ alt, url, seccion: null }),
                        });
                        console.log("Guardado en galerÃ­a:", galeriaRes.status);
                    }

                    console.log("Payload final:", {
                        titulo,
                        lugar,
                        fecha,
                        hora,
                        imagenUrl,
                    });

                    const res = await fetch(`/api/eventos/${id}`, {
                        method: "PUT",
                        headers: { "Content-Type": "application/json" },
                        credentials: "include",
                        body: JSON.stringify({
                            titulo,
                            lugar,
                            fecha,
                            hora,
                            imagenUrl,
                        }),
                    });

                    console.log("PUT status:", res.status);

                    if (res.ok) {
                        const data = await res.json();
                        console.log("PUT response:", data);
                        window.location.href = "/admin";
                    } else {
                        const data = await res.json().catch(() => ({}));
                        console.error("PUT error:", data);
                        errorMsg.textContent = data.error || "Error al guardar";
                        errorMsg.style.display = "block";
                    }
                } catch (err) {
                    console.error("Error en submit:", err);
                    errorMsg.textContent =
                        err instanceof Error
                            ? err.message
                            : "Error de conexiÃ³n";
                    errorMsg.style.display = "block";
                } finally {
                    submitBtn.removeAttribute("disabled");
                    submitBtn.textContent = "Guardar cambios";
                }
            });
        }
